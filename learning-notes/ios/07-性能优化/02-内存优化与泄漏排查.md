## 1. 循环引用检测

```swift
// ❌ Delegate 循环引用
class ParentVC: UIViewController {
    let child = ChildVC()
    override func viewDidLoad() {
        child.delegate = self  // child 强引用 parent
    }
}

// ✅ weak delegate
protocol ChildDelegate: AnyObject { func childDidFinish() }
class ChildVC: UIViewController {
    weak var delegate: ChildDelegate?  // weak 打破循环
}

// ❌ 闭包捕获 self
class ViewModel {
    var onUpdate: (() -> Void)?
    func setup() {
        onUpdate = { self.refresh() }  // 循环引用
    }
}

// ✅ [weak self] 或 [unowned self]
onUpdate = { [weak self] in self?.refresh() }
```

```swift
// 常见泄漏场景
class NetworkManager {
    // ❌ NotificationCenter 未移除
    func setup() {
        NotificationCenter.default.addObserver(self, selector: #selector(handle),
                                                name: .someNotification, object: nil)
    }
    // ✅ 使用闭包 API（自动管理）
    private var observer: Any?
    func setup() {
        observer = NotificationCenter.default.addObserver(forName: .someNotification,
                                                           object: nil, queue: .main) { [weak self] _ in
            self?.handle()
        }
    }
    deinit { observer.map { NotificationCenter.default.removeObserver($0) } }
}
```

## 3. Combine 订阅泄漏

```swift
class ViewModel: ObservableObject {
    private var cancellables = Set<AnyCancellable>()

    // ✅ store(in:) 自动管理生命周期
    func bind() {
        publisher
            .sink { [weak self] value in self?.handle(value) }
            .store(in: &cancellables)
    }

    // ❌ 忘记存储 cancellable → 订阅立即取消
    func badBind() {
        publisher.sink { value in }  // 没有 store → 立即释放
    }
}
```

## 4. 图片内存优化

```swift
// ❌ 直接加载大图
let image = UIImage(named: "huge_photo")  // 可能占用几十MB

// ✅ 降采样（Downsampling）
func downsample(imageAt url: URL, to pointSize: CGSize, scale: CGFloat) -> UIImage? {
    let maxDimension = max(pointSize.width, pointSize.height) * scale
    let options: [CFString: Any] = [
        kCGImageSourceCreateThumbnailFromImageAlways: true,
        kCGImageSourceShouldCacheImmediately: true,
        kCGImageSourceCreateThumbnailWithTransform: true,
        kCGImageSourceThumbnailMaxPixelSize: maxDimension
    ]
    guard let source = CGImageSourceCreateWithURL(url as CFURL, nil),
          let cgImage = CGImageSourceCreateThumbnailAtIndex(source, 0, options as CFDictionary) else {
        return nil
    }
    return UIImage(cgImage: cgImage)
}

// ✅ 使用 UIImage 的系统缓存
let image = UIImage(named: "icon")  // 系统自动缓存
let image2 = UIImage(contentsOfFile: path)  // 不缓存，适合大图
```

## 5. Autorelease Pool

```swift
// 循环中创建大量临时对象
func processLargeDataset() {
    for i in 0..<100000 {
        autoreleasepool {
            let data = generateData(index: i)
            processData(data)
            // data 在每次循环结束时释放，避免内存峰值
        }
    }
}
```

## 6. 内存警告处理

```swift
class ImageCache {
    static let shared = ImageCache()
    private var cache = NSCache<NSString, UIImage>()

    init() {
        cache.totalCostLimit = 100 * 1024 * 1024  // 100MB 上限
        // 监听内存警告
        NotificationCenter.default.addObserver(forName: UIApplication.didReceiveMemoryWarningNotification,
                                                object: nil, queue: .main) { [weak self] _ in
            self?.cache.removeAllObjects()
        }
    }
}
```

## 7. Swift 6.2 内存安全与零分配原语

<!-- version-check: Swift 6.2 strict memory safety, InlineArray, Span, checked 2026-07-10 -->
<!-- 修复于 2026-07-10: InlineArray 重复值初始化示例语法有误（[0, repeat: 0] 不是合法 Swift 语法），改为官方 .init(repeating:) 初始化器 -->

> 🔄 更新于 2026-05-06

Swift 6.2 引入了一系列面向系统编程的内存原语，可以在高性能路径上消除堆分配和引用计数开销，同时保持编译器检查的安全性。相比 `UnsafeBufferPointer` 和手写 tuple 存储，这些新类型是更好的选择。

### 7.1 InlineArray：栈分配的固定大小数组

`InlineArray<N, Element>` 把所有元素内联存储在值本身（通常位于栈），零堆分配、零引用计数，适合大小在编译期可确定的热路径数据。

```swift
// ❌ 数组频繁创建导致堆分配
func render(pixels: [Int]) -> [Int] {  // [Int] 堆分配 + ARC
    pixels.map { $0 * 2 }
}

// ✅ 栈分配的 InlineArray，零堆分配
func render(pixels: InlineArray<1024, Int>) -> InlineArray<1024, Int> {
    var result: [1024 of Int] = .init(repeating: 0)  // Swift 6.2 类型字面量语法 [count of Element]
    for i in 0..<1024 {
        result[i] = pixels[i] * 2
    }
    return result
}

// 也可以作为存储属性使用，避免 ContiguousArray 的间接引用
struct Histogram {
    var buckets: [256 of Int] = .init(repeating: 0)  // 直接内联 2KB 数据
}
```

适用场景：图像像素缓冲、音频样本窗口、嵌入式场景固定大小的数据包。

### 7.2 Span / RawSpan：零成本的安全切片

`Span<T>` 和 `RawSpan` 提供对连续内存的边界检查访问，但没有 `UnsafeBufferPointer` 的生命周期陷阱。它们是 non-escapable，编译器保证不会在底层存储释放后被使用。

```swift
// ❌ UnsafeBufferPointer 的经典陷阱
func parseHeader(data: Data) -> UnsafeBufferPointer<UInt8> {
    data.withUnsafeBytes { buffer in
        return buffer.bindMemory(to: UInt8.self)  // 危险：data 释放后 buffer 悬挂
    }
}

// ✅ Span 不可逃逸，编译器阻止悬挂引用
func checksum(of bytes: Span<UInt8>) -> UInt32 {
    var sum: UInt32 = 0
    for i in 0..<bytes.count {
        sum = sum &+ UInt32(bytes[i])  // 编译期检查越界
    }
    return sum
}

// 调用
let data: [UInt8] = [0x01, 0x02, 0x03]
let result = data.withSpan { span in
    checksum(of: span)  // span 只在闭包内有效
}
```

SE-0525（2026-04-16 通过）进一步为 `RawSpan` 添加了安全的类型化加载 API（`loadUnaligned(fromByteOffset:as:)` 等），可以在协议解析等场景完全替代 `UnsafeRawPointer`。

> 来源：[Swift 6.2 Released](https://www.infoq.com/news/2025/09/swift-6-2-released/) ｜ [SE-0525 Safe loading API for RawSpan](https://forums.swift.org/t/accepted-with-modifications-se-0525-safe-loading-api-for-rawspan/86329)

### 7.3 严格内存安全模式

Swift 6.2 新增 opt-in 的 strict memory safety 模式，编译器会标记所有使用 `Unsafe*` 指针、C 指针互操作等不安全构造的代码，要求显式确认。这让代码库可以逐步消除不安全路径。

```swift
// Package.swift 中开启
.target(
    name: "MyLib",
    swiftSettings: [
        .enableExperimentalFeature("StrictMemorySafety")
    ]
)
```

开启后，像 `UnsafeBufferPointer`、`Unmanaged`、`OpaquePointer` 的使用都会触发警告，强制迁移到 `Span` / `InlineArray` 或添加 `@unsafe` 显式标注。

> 来源：[Swift 6.2 Strict Memory Safety](https://mjtsai.com/blog/2025/10/29/swift-6-2/) ｜ [Safely Mixing Swift and C/C++](https://swift.org/documentation/cxx-interop/safe-interop/)

### 7.4 选型指南

| 场景 | 推荐类型 | 原因 |
| ---- | ------ | ---- |
| 动态大小数组 | `Array` / `ContiguousArray` | 通用场景，堆分配但 API 最完善 |
| 编译期已知大小的数组 | `InlineArray<N, T>` | 栈分配零堆开销，无 ARC |
| 临时切片读取 | `Span<T>` | 边界检查、不可逃逸、零成本 |
| 字节协议解析 | `RawSpan` + SE-0525 API | 类型化加载，替代 `UnsafeRawPointer` |
| C 互操作 | `withUnsafePointer` + `@unsafe` | 显式标注不安全边界 |
| 向量计算 | `SIMD` + `InlineArray` | 组合使用，利用硬件向量指令 |

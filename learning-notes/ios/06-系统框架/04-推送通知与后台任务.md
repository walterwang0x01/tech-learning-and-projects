## 1. 请求通知权限

```swift
import UserNotifications

func requestNotificationPermission() {
    UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge]) { granted, error in
        if granted { print("通知权限已授权") }
        DispatchQueue.main.async {
            UIApplication.shared.registerForRemoteNotifications()
        }
    }
}
```

## 2. 本地通知

```swift
func scheduleLocalNotification() {
    let content = UNMutableNotificationContent()
    content.title = "提醒"
    content.body = "该喝水了"
    content.sound = .default
    content.badge = 1

    // 时间触发
    let trigger = UNTimeIntervalNotificationTrigger(timeInterval: 60, repeats: false)

    // 日期触发
    var dateComponents = DateComponents()
    dateComponents.hour = 9
    dateComponents.minute = 0
    let dateTrigger = UNCalendarNotificationTrigger(dateMatching: dateComponents, repeats: true)

    let request = UNNotificationRequest(identifier: "water-reminder", content: content, trigger: trigger)
    UNUserNotificationCenter.current().add(request)
}

// 取消通知
UNUserNotificationCenter.current().removePendingNotificationRequests(withIdentifiers: ["water-reminder"])
```

## 4. 处理通知响应

```swift
class NotificationDelegate: NSObject, UNUserNotificationCenterDelegate {
    // 前台收到通知
    func userNotificationCenter(_ center: UNUserNotificationCenter,
                                willPresent notification: UNNotification,
                                withCompletionHandler completionHandler: @escaping (UNNotificationPresentationOptions) -> Void) {
        completionHandler([.banner, .sound, .badge])
    }

    // 用户点击通知
    func userNotificationCenter(_ center: UNUserNotificationCenter,
                                didReceive response: UNNotificationResponse,
                                withCompletionHandler completionHandler: @escaping () -> Void) {
        let userInfo = response.notification.request.content.userInfo
        handleNotification(userInfo)
        completionHandler()
    }
}
```

## 5. BGTaskScheduler 后台任务

```swift
import BackgroundTasks

// 注册后台任务（AppDelegate didFinishLaunching）
BGTaskScheduler.shared.register(forTaskWithIdentifier: "com.app.refresh", using: nil) { task in
    handleAppRefresh(task: task as! BGAppRefreshTask)
}

// 调度任务
func scheduleAppRefresh() {
    let request = BGAppRefreshTaskRequest(identifier: "com.app.refresh")
    request.earliestBeginDate = Date(timeIntervalSinceNow: 15 * 60)  // 最早15分钟后
    try? BGTaskScheduler.shared.submit(request)
}

// 处理后台任务
func handleAppRefresh(task: BGAppRefreshTask) {
    scheduleAppRefresh()  // 调度下一次

    let operation = RefreshOperation()
    task.expirationHandler = { operation.cancel() }

    operation.completionBlock = {
        task.setTaskCompleted(success: !operation.isCancelled)
    }
    OperationQueue().addOperation(operation)
}
```

## 7. iOS 26 推送通知与后台任务新特性

### 7.1 推送通知变化

iOS 26 对推送通知引入了多项改进，重点在智能分类和优先级管理：

**AI 驱动的通知智能排序**：系统自动对通知进行优先级排序，重要通知（如时间敏感的交易消息）会优先展示，低优先级通知可能被折叠到摘要中。

**通知屏幕闪烁辅助功能**（iOS 26.2）：新增屏幕闪烁提示，增强无障碍访问体验。

**UIScene 生命周期迁移**：iOS 26 开始在控制台输出警告，要求应用采用 `UISceneDelegate` 生命周期。iOS 27 将强制要求。这影响推送通知的处理方式——通知回调需要在 Scene 级别处理。

来源：[OneSignal iOS 26 Recap](https://onesignal.com/blog/wwdc-2025-new-release-recap/)、[Courier UIScene 迁移指南](https://courier.com/blog/ios-26-push-notification-changes-uiscene-requirment-ios-27)

```swift
// iOS 26+ 推荐：在 SceneDelegate 中处理通知
class SceneDelegate: UIResponder, UIWindowSceneDelegate {
    var window: UIWindow?

    func scene(_ scene: UIScene, willConnectTo session: UISceneSession,
               options connectionOptions: UIScene.ConnectionOptions) {
        // 处理冷启动时的通知
        if let response = connectionOptions.notificationResponse {
            handleNotification(response.notification.request.content.userInfo)
        }
    }
}

// SwiftUI 中处理通知
@main
struct MyApp: App {
    @UIApplicationDelegateAdaptor(AppDelegate.self) var appDelegate

    var body: some Scene {
        WindowGroup {
            ContentView()
                .onOpenURL { url in
                    // 处理通知中的深链接
                }
        }
    }
}
```

### 7.2 BGContinuedProcessingTask（iOS 26 新增）

iOS 26 引入 `BGContinuedProcessingTask`，允许应用在从前台切换到后台时继续完成用户发起的长时间任务（如视频导出、文件处理），无需用户保持应用在前台。

来源：[WWDC25 Session: Finish tasks in the background](https://developer.apple.com/videos/play/wwdc2025/227/)

```swift
import BackgroundTasks

// 注册后台继续处理任务
BGTaskScheduler.shared.register(
    forTaskWithIdentifier: "com.app.videoExport",
    using: nil
) { task in
    guard let continuedTask = task as? BGContinuedProcessingTask else { return }
    handleContinuedProcessing(task: continuedTask)
}

// 在前台开始任务，切换到后台时自动继续
func startVideoExport(videoURL: URL) {
    let request = BGContinuedProcessingTaskRequest(
        identifier: "com.app.videoExport"
    )
    // 系统会在应用进入后台时自动接管任务
    try? BGTaskScheduler.shared.submit(request)

    // 开始实际的导出工作
    exportVideo(from: videoURL) { progress in
        // 更新进度（前台和后台都会执行）
        updateProgress(progress)
    }
}

func handleContinuedProcessing(task: BGContinuedProcessingTask) {
    task.expirationHandler = {
        // 系统要求停止时的清理逻辑
        savePartialProgress()
    }

    // 任务完成后通知系统
    task.setTaskCompleted(success: true)
}
```

**BGContinuedProcessingTask vs 其他后台 API 对比**：

| API | 适用场景 | 运行时机 | iOS 版本 |
|-----|---------|---------|---------|
| `BGAppRefreshTask` | 定期数据刷新 | 系统调度（不可预测） | iOS 13+ |
| `BGProcessingTask` | 大型数据处理 | 设备充电/空闲时 | iOS 13+ |
| `BGContinuedProcessingTask` | 用户发起的长任务 | 前台→后台过渡时 | **iOS 26+** |
| `beginBackgroundTask` | 短时后台完成 | 进入后台后约 30 秒 | iOS 4+ |
| Background URLSession | 大文件下载/上传 | 系统管理 | iOS 7+ |

### 7.3 Background GPU Access（iOS 26 新增）

iOS 26 还引入了后台 GPU 访问能力，配合 `BGContinuedProcessingTask` 使用，允许视频处理、图像渲染等 GPU 密集型任务在后台继续执行。需要在 Xcode 中启用 "Background GPU Access" capability。

### 8.1 APNs Broadcast Push Notifications

APNs 新增**广播推送（Broadcast Push）**能力，让应用可以**通过单个推送请求**向**所有订阅的 Live Activities 用户**发送更新，无需逐个 device token 推送。这是面向「赛事比分、新闻直播、股价播报」等大规模 Live Activities 场景的关键基础设施。

来源：[Apple Notifications Overview](https://developer.apple.com/notifications/)、[Push Notifications Console](https://developers.apple.com/notifications/push-notifications-console/)

**与传统 Live Activities 推送对比**：

| 维度 | 传统 ActivityKit Push | APNs Broadcast Push |
|------|----------------------|---------------------|
| 推送目标 | 单个 device push token | 频道（channel）订阅者 |
| 服务端复杂度 | 维护 token 列表 + 批量并发 | 单次请求 |
| 规模上限 | 受限于服务端并发能力 | 数百万订阅者 |
| 典型场景 | 个人订单、私聊消息 | 赛事比分、新闻、股价 |

**典型订阅模式**：

```swift
import ActivityKit

// 1. 客户端启动 Live Activity 并订阅频道
let attributes = MatchAttributes(matchId: "nba-2026-finals-g4")
let initialState = MatchAttributes.ContentState(
    homeScore: 0,
    awayScore: 0
)

// 启动 Activity（订阅频道，由 attributes 派生）
let activity = try Activity<MatchAttributes>.request(
    attributes: attributes,
    content: .init(state: initialState, staleDate: nil),
    pushType: .channel  // iOS 26+ 新增 channel 推送类型
)

// 2. 服务端：单次请求广播给所有订阅者
// POST /3/broadcasts/<channel-id>
// Headers:
//   apns-topic: <bundle-id>.push-type.liveactivity
//   apns-push-type: liveactivity
// Body:
//   {
//     "aps": { "timestamp": ..., "event": "update",
//              "content-state": { "homeScore": 96, "awayScore": 92 } }
//   }
```

### 8.2 iOS 26.5 Live Activities 第三方设备转发（EU DMA）

iOS 26.5（**RC 2026-05-04，GA 2026-05-11**）为符合欧盟数字市场法案（DMA）的可穿戴设备生态新增了 **Live Activities 与通知转发**能力，让第三方智能手表、耳机可以像 AirPods/Apple Watch 一样接收 iPhone 通知。

来源：[Apple Releases iOS 26.5](https://www.macrumors.com/2026/05/11/apple-releases-ios-26-5/)、[iOS 26.5 Beta 1 概览](https://www.macrumors.com/2026/03/30/everything-new-in-ios-26-5-beta-1/)、[Apple Privacy Rules for Forwarding](https://macrumors.com/2026/03/31/apple-sets-privacy-rules-live-activities-alerts/)

**关键能力**：

- **Proximity Pairing**：第三方耳机靠近 iPhone 即可一键配对（AirPods 风格的弹窗）
- **Notification Forwarding**：iPhone 通知镜像到第三方手表
- **Live Activities Forwarding**：Dynamic Island 内容同步到第三方可穿戴设备

**Apple 强制的隐私规则**（开发者协议 2026-03 修订）：

| 规则 | 要求 |
|------|------|
| 数据存储位置 | **不得**在服务器、云端、远程设备上存储转发的通知数据 |
| 解密位置 | **必须**在配件本身解密，传输链路上任何节点都不得解密 |
| 数据用途 | 仅可用于在配件上展示，不得用于分析、广告等其他目的 |
| 适用范围 | 当前仅限 EU 区域（DMA 合规） |

**对开发者的影响**：

- 如果你的应用使用 Live Activities，第三方设备生态会被动扩展，无需额外代码
- 如果你开发**配件相关**应用（如自定义手表 SDK），必须在 Info.plist 声明 `NSLiveActivitiesForwardingUsageDescription` 并实现端到端加密的数据通道
- 自带服务端的应用需评估**端到端加密**改造成本：传输层 TLS 不够，必须确保 payload 在 APNs 网关之外仍是加密的

### 8.3 关键实践建议

1. **Broadcast Push 适用边界**：当订阅者超过 1 万且更新频率高（秒级）时显著优于逐 token 推送，但**不能携带个性化数据**（所有订阅者收到相同内容）。个性化场景仍需传统 token 推送。
2. **频道 ID 设计**：建议按业务实体（比赛、航班、订单组）划分频道 ID，避免单一频道订阅者过载。
3. **EU 区域适配**：如果应用在欧盟分发，提前测试 iOS 26.5 上的转发行为，确保 Live Activity ContentState 不包含敏感数据（因为可能镜像到第三方设备）。
4. **WWDC26 跟进**：[WWDC26](https://developer.apple.com/wwdc26/) 将于 **2026-06-08** 召开，预计会发布 iOS 27 + 新的 Live Activities/Push 能力，建议在 6 月前完成 iOS 26 系列适配，为 iOS 27 留出窗口。

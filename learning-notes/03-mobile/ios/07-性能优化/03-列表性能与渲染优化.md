## 1. Cell 复用机制

```swift
// ✅ 正确注册和复用
tableView.register(UserCell.self, forCellReuseIdentifier: "UserCell")

func tableView(_ tableView: UITableView, cellForRowAt indexPath: IndexPath) -> UITableViewCell {
    let cell = tableView.dequeueReusableCell(withIdentifier: "UserCell", for: indexPath) as! UserCell
    cell.configure(with: users[indexPath.row])
    return cell
}

// ✅ Cell 中重置状态，避免复用导致数据错乱
class UserCell: UITableViewCell {
    override func prepareForReuse() {
        super.prepareForReuse()
        avatarImageView.image = nil
        nameLabel.text = nil
        downloadTask?.cancel()
    }
}
```

## 2. 预计算 Cell 高度

```swift
// ❌ 每次都动态计算高度（卡顿）
func tableView(_ tableView: UITableView, heightForRowAt indexPath: IndexPath) -> CGFloat {
    return UITableView.automaticDimension  // 复杂布局时可能卡顿
}

// ✅ 缓存高度
private var heightCache: [IndexPath: CGFloat] = [:]

func tableView(_ tableView: UITableView, estimatedHeightForRowAt indexPath: IndexPath) -> CGFloat {
    return heightCache[indexPath] ?? 80  // 预估高度
}

func tableView(_ tableView: UITableView, willDisplay cell: UITableViewCell, forRowAt indexPath: IndexPath) {
    heightCache[indexPath] = cell.frame.height  // 缓存实际高度
}
```

## 3. 异步图片加载

```swift
class AsyncImageCell: UITableViewCell {
    private var currentURL: URL?

    func configure(with imageURL: URL) {
        currentURL = imageURL
        imageView?.image = UIImage(named: "placeholder")

        Task {
            guard let (data, _) = try? await URLSession.shared.data(from: imageURL),
                  currentURL == imageURL else { return }  // 防止复用错乱
            let image = UIImage(data: data)
            await MainActor.run {
                guard self.currentURL == imageURL else { return }
                self.imageView?.image = image
            }
        }
    }
}
```

## 4. 离屏渲染优化

```swift
// ❌ 触发离屏渲染的操作
view.layer.cornerRadius = 10
view.layer.masksToBounds = true  // 配合 cornerRadius 触发离屏渲染
view.layer.shadowPath = nil      // 无 shadowPath 触发离屏渲染

// ✅ 避免离屏渲染
// 方案1：指定 shadowPath
view.layer.shadowPath = UIBezierPath(roundedRect: view.bounds, cornerRadius: 10).cgPath

// 方案2：使用 CAShapeLayer 做圆角
let maskLayer = CAShapeLayer()
maskLayer.path = UIBezierPath(roundedRect: view.bounds, cornerRadius: 10).cgPath
view.layer.mask = maskLayer

// 方案3：iOS 13+ 指定圆角位置（不触发离屏渲染）
view.layer.cornerRadius = 10
view.layer.cornerCurve = .continuous
view.layer.maskedCorners = [.layerMinXMinYCorner, .layerMaxXMinYCorner]
// 仅设置 cornerRadius 不设置 masksToBounds 不会触发离屏渲染
```

## 5. 预加载与分页

```swift
// UITableViewDataSourcePrefetching
class ListVC: UIViewController, UITableViewDataSourcePrefetching {
    func tableView(_ tableView: UITableView, prefetchRowsAt indexPaths: [IndexPath]) {
        for indexPath in indexPaths {
            let url = items[indexPath.row].imageURL
            ImagePrefetcher.shared.prefetch(url: url)
        }
    }

    func tableView(_ tableView: UITableView, cancelPrefetchingForRowsAt indexPaths: [IndexPath]) {
        for indexPath in indexPaths {
            ImagePrefetcher.shared.cancel(url: items[indexPath.row].imageURL)
        }
    }
}

// 分页加载
func scrollViewDidScroll(_ scrollView: UIScrollView) {
    let offsetY = scrollView.contentOffset.y
    let contentHeight = scrollView.contentSize.height
    let frameHeight = scrollView.frame.height

    if offsetY > contentHeight - frameHeight * 2 {
        loadNextPage()
    }
}
```

## 7. Xcode 26 SwiftUI Instrument

<!-- version-check: SwiftUI Instrument Xcode 26, checked 2026-05-06 -->

> 🔄 更新于 2026-05-06

Xcode 26 新增的 SwiftUI Instrument 可以在时间线上直接看到每次 view body 执行的耗时、触发来源和 identity 变化，是定位 SwiftUI 列表卡顿的首选工具。

### 7.1 核心能力

- Long View Body Updates：标记执行超过目标帧时长的 view body，用橙/红色柱状图直观展示
- Cause & Effect Graph：展示状态变化 → view 更新的因果链，快速找到触发刷新的 source
- Identity Tracking：识别不必要的 view 重建（如 `ForEach` 中 id 频繁变化导致整个 cell 被重建）
- View Body 时间线：按时间轴显示每个 view 的更新分布，结合滚动事件分析卡顿位置

### 7.2 常见的 SwiftUI 列表性能问题

```swift
// ❌ ForEach 没有稳定 id，导致滚动时 cell 被重建
struct BadList: View {
    let items: [Item]
    var body: some View {
        List {
            ForEach(items, id: \.self) { item in  // Item 不是 Hashable 稳定实现会踩坑
                ItemRow(item: item)
            }
        }
    }
}

// ✅ 使用稳定 id（如业务主键）
struct GoodList: View {
    let items: [Item]
    var body: some View {
        List {
            ForEach(items, id: \.id) { item in
                ItemRow(item: item)
            }
        }
    }
}

// ❌ 大型 @Observable 对象作为环境值，任何字段变化都会刷新所有 cell
@Observable
class Store {
    var filter: String = ""
    var items: [Item] = []
    var unrelatedSettings: Settings = .init()  // 变化也会导致列表重建
}

// ✅ 用 @State 局部 scope 或拆分多个 Observable
@Observable class ItemStore { var items: [Item] = [] }
@Observable class SettingsStore { var unrelatedSettings: Settings = .init() }
```

### 7.3 工作流程

1. Xcode → Product → Profile → SwiftUI 模板
2. 运行应用，执行要分析的交互（如快速滚动）
3. 在时间线上选择一个 long update 区间（橙/红色）
4. 查看 Cause & Effect Graph，找到触发 body 更新的状态变化源
5. 如果是不必要的更新，拆分 Observable / 使用 `Equatable` 视图 / 把昂贵计算挪到 `.task` 或 background actor

### 7.4 与传统工具的配合

| 工具 | 适用场景 |
| ---- | ------ |
| SwiftUI Instrument（Xcode 26） | SwiftUI view body 耗时、重建原因 |
| Time Profiler | 非 SwiftUI 代码路径的 CPU 热点 |
| Allocations | Cell 数据的对象分配分析 |
| Hitches | 直接测量丢帧次数和原因 |
| Processor Trace（Apple Silicon） | 指令级别追踪，用于理解 JIT 或 LLDB 无法解释的 case |

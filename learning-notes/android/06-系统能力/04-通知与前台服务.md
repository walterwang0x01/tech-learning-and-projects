## 1. NotificationChannel（Android 8.0+）

```kotlin
class MyApp : Application() {
    override fun onCreate() {
        super.onCreate()
        createNotificationChannels()
    }

    private fun createNotificationChannels() {
        val channels = listOf(
            NotificationChannel("messages", "消息通知", NotificationManager.IMPORTANCE_HIGH).apply {
                description = "聊天消息通知"
                enableVibration(true)
                enableLights(true)
            },
            NotificationChannel("updates", "更新通知", NotificationManager.IMPORTANCE_DEFAULT).apply {
                description = "应用更新提醒"
            },
            NotificationChannel("background", "后台任务", NotificationManager.IMPORTANCE_LOW).apply {
                description = "后台任务进度"
            }
        )
        val manager = getSystemService(NotificationManager::class.java)
        manager.createNotificationChannels(channels)
    }
}
```

## 2. 发送通知

```kotlin
class NotificationHelper(private val context: Context) {

    // 基本通知
    fun showSimpleNotification(title: String, content: String) {
        val intent = Intent(context, MainActivity::class.java)
        val pendingIntent = PendingIntent.getActivity(
            context, 0, intent, PendingIntent.FLAG_IMMUTABLE
        )

        val notification = NotificationCompat.Builder(context, "messages")
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(title)
            .setContentText(content)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setContentIntent(pendingIntent)
            .setAutoCancel(true)
            .build()

        NotificationManagerCompat.from(context).notify(1, notification)
    }

    // 大文本通知
    fun showBigTextNotification(title: String, longText: String) {
        val notification = NotificationCompat.Builder(context, "messages")
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(title)
            .setStyle(NotificationCompat.BigTextStyle().bigText(longText))
            .build()

        NotificationManagerCompat.from(context).notify(2, notification)
    }

    // 进度通知
    fun showProgressNotification(title: String, progress: Int) {
        val notification = NotificationCompat.Builder(context, "background")
            .setSmallIcon(R.drawable.ic_download)
            .setContentTitle(title)
            .setProgress(100, progress, false)
            .setOngoing(true)
            .build()

        NotificationManagerCompat.from(context).notify(3, notification)
    }

    // 带操作按钮的通知
    fun showActionNotification(title: String, content: String) {
        val replyIntent = PendingIntent.getBroadcast(
            context, 0, Intent("REPLY_ACTION"), PendingIntent.FLAG_IMMUTABLE
        )
        val notification = NotificationCompat.Builder(context, "messages")
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(title)
            .setContentText(content)
            .addAction(R.drawable.ic_reply, "回复", replyIntent)
            .addAction(R.drawable.ic_dismiss, "忽略", null)
            .build()

        NotificationManagerCompat.from(context).notify(4, notification)
    }
}
```

## 3. 前台服务通知

```kotlin
class UploadForegroundService : Service() {
    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val notification = createNotification("准备上传...", 0)

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.UPSIDE_DOWN_CAKE) {
            startForeground(100, notification, ServiceInfo.FOREGROUND_SERVICE_TYPE_DATA_SYNC)
        } else {
            startForeground(100, notification)
        }

        startUpload()
        return START_NOT_STICKY
    }

    private fun createNotification(text: String, progress: Int): Notification {
        return NotificationCompat.Builder(this, "background")
            .setSmallIcon(R.drawable.ic_upload)
            .setContentTitle("文件上传")
            .setContentText(text)
            .setProgress(100, progress, progress == 0)
            .setOngoing(true)
            .build()
    }

    private fun updateProgress(progress: Int) {
        val notification = createNotification("上传中 $progress%", progress)
        NotificationManagerCompat.from(this).notify(100, notification)
    }

    override fun onBind(intent: Intent?): IBinder? = null
}

// AndroidManifest.xml
// <uses-permission android:name="android.permission.FOREGROUND_SERVICE" />
// <uses-permission android:name="android.permission.FOREGROUND_SERVICE_DATA_SYNC" />
// <uses-permission android:name="android.permission.POST_NOTIFICATIONS" />
```

## 4. 通知权限（Android 13+）

```kotlin
@Composable
fun NotificationPermissionRequest() {
    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
        val permissionState = rememberPermissionState(
            Manifest.permission.POST_NOTIFICATIONS
        )
        LaunchedEffect(Unit) {
            if (!permissionState.status.isGranted) {
                permissionState.launchPermissionRequest()
            }
        }
    }
}
```

## 5. Android 16 (API 36) 通知与前台服务变化

<!-- version-check: Android 16 API 36, Notification + FGS changes, checked 2026-04-23 -->

Android 16 对通知和前台服务引入了多项重要变化，面向 targetSdk 36 的应用必须适配。来源：[Android 16 Behavior Changes](https://developer.android.com/about/versions/16/behavior-changes-16)、[FGS Changes](https://developer.android.com/develop/background-work/services/fgs/changes)

### 5.1 通知新特性

```
Android 16 通知变化：
├─ 通知自动分组 — 未设置 summary 的通知自动分组
├─ AI 通知摘要 — 群聊/长消息自动生成摘要
├─ Progress-centric 通知 — 进度型通知新样式
├─ Compact 通知 — 紧凑型通知布局
└─ Notification Organizer — 系统级通知管理
```

### 5.2 前台服务限制加强

```kotlin
// Android 16 前台服务要求：
// 1. 必须声明 foregroundServiceType（更严格）
// 2. 类型特定权限必须请求
// 3. 后台启动需要豁免条件
// 4. 时间配额更严格

// AndroidManifest.xml
// <service
//     android:name=".LocationService"
//     android:foregroundServiceType="location"
//     android:exported="false" />

// 启动前台服务（API 36）
class LocationService : Service() {
    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val notification = NotificationCompat.Builder(this, "location")
            .setContentTitle("位置追踪")
            .setContentText("正在获取位置...")
            .setSmallIcon(R.drawable.ic_location)
            .setOngoing(true) // API 36 推荐：非交互通知
            .build()

        // API 34+ 必须指定 foregroundServiceType
        ServiceCompat.startForeground(
            this,
            NOTIFICATION_ID,
            notification,
            ServiceInfo.FOREGROUND_SERVICE_TYPE_LOCATION
        )
        return START_STICKY
    }
}
```

### 5.3 推荐替代方案

```
前台服务 vs 替代方案（2026）：
├─ 短期后台任务 → WorkManager（推荐）
├─ 精确定时任务 → AlarmManager + WorkManager
├─ 数据同步 → WorkManager + PeriodicWorkRequest
├─ 位置追踪 → 前台服务（仍需要）
├─ 音乐播放 → 前台服务（仍需要）
└─ 文件下载 → DownloadManager（推荐）
```

## 6. Android 17 (API 37) 通知与前台服务变化

<!-- version-check: Android 17 API 37 stable, App Bubbles, Semantic Notification, Kotlin 2.4.0, checked 2026-07-08 -->

Android 17 稳定版（2026-06）在通知、气泡多任务和后台音频方面引入多项开发者可见变化。来源：[Android 17 Features](https://developer.android.com/about/versions/17/features)、[Android 17 Release Notes](https://developer.android.com/about/versions/17/release-notes)、[Behavior changes: all apps](https://developer.android.com/about/versions/17/behavior-changes-all)

### 6.1 App Bubbles（系统级悬浮气泡）

```kotlin
// 聊天/导航类 App 可声明 BubbleMetadata，系统提供类似 PiP 的悬浮窗口
val bubbleIntent = PendingIntent.getActivity(
    context, 0, Intent(context, ChatActivity::class.java),
    PendingIntent.FLAG_IMMUTABLE
)
val bubbleMetadata = Notification.BubbleMetadata.Builder(
    shortcutId,
    Icon.createWithResource(context, R.drawable.ic_bubble),
    bubbleIntent
).setAutoExpandBubble(true)
 .setSuppressNotification(true)
 .build()

NotificationCompat.Builder(context, CHANNEL_ID)
    .setBubbleMetadata(bubbleMetadata)
    .setCategory(Notification.CATEGORY_MESSAGE)
    .build()
```

来源：[Android 17 用户向新特性 — floating Bubbles](https://blog.google/products-and-platforms/platforms/android/android-17-features/)

### 6.2 Live Update 语义着色 API

```kotlin
// Android 17 为进度型 / Live Update 通知提供语义色标注
val ssb = SpannableStringBuilder()
    .append("状态: ")
    .append("安全", Notification.createSemanticStyleAnnotation(
        Notification.SEMANTIC_STYLE_SAFE
    ), 0)

Notification.Builder(context, channelId)
    .setContentText(ssb)
    .setOngoing(true)
    .setRequestPromotedOngoing(true)  // 提升为 Live Update
    .build()
```

语义色：`SEMANTIC_STYLE_SAFE`（绿）、`CAUTION`（橙）、`DANGER`（红）、`INFO`（蓝）。来源：[Android 17 Features — Semantic color API](https://developer.android.com/about/versions/17/features#live-update-semantic-color-api)

### 6.3 自定义通知视图尺寸限制

```
Android 17 运行时变化（所有应用）：
├─ 自定义 RemoteViews 通知强制执行严格尺寸上限
├─ 超限通知可能被系统裁剪或拒绝渲染
└─ 建议：复杂 UI 改用标准样式 + Live Update / ProgressStyle
```

### 6.4 后台音频加固（targetSdk 37+）

```kotlin
// Android 17 对后台音频播放、焦点请求、音量 API 加强生命周期校验
// 非前台 / 无有效 FGS 时调用可能静默失败（fail silently）
// 闹钟音频已豁免；while-in-use FGS 按 targetSdk 分阶段强制

// ✅ 音乐/导航类仍须合规前台服务 + foregroundServiceType
ServiceCompat.startForeground(
    this, NOTIFICATION_ID, notification,
    ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PLAYBACK
)
```

来源：[Android 17 Release Notes — Background audio hardening](https://developer.android.com/about/versions/17/release-notes)

### 6.5 Assistant 独立音量流

Android 17 为 `USAGE_ASSISTANT` 音频引入独立 Assistant 音量流，与媒体音量解耦；支持 `MODE_ASSISTANT_CONVERSATION` 的 Assistant App 可在会话外独立调节音量。来源：[Android 17 Features — Dedicated Assistant volume stream](https://developer.android.com/about/versions/17/features#dedicated-assistant-volume-stream)

### 6.6 适配优先级（2026 Q3）

| 优先级 | 项目 | 风险 |
|--------|------|------|
| ⭐️⭐️⭐️ | 后台音频 + FGS 生命周期合规 | 静默播放失败 |
| ⭐️⭐️ | BubbleMetadata 适配（聊天/协作类） | 多任务体验落后 |
| ⭐️⭐️ | 自定义通知 RemoteViews 瘦身 | 渲染被裁切 |
| ⭐️ | Live Update 语义着色 | 进度通知可读性 |

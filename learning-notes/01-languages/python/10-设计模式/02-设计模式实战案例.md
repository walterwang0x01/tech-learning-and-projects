## 1. 观察者模式 - 事件系统

### 1.1 实现事件发布订阅

```python
class Event:
    def __init__(self, name, data=None):
        self.name = name
        self.data = data

class EventEmitter:
    def __init__(self):
        self._listeners = {}
    
    def on(self, event_name, callback):
        if event_name not in self._listeners:
            self._listeners[event_name] = []
        self._listeners[event_name].append(callback)
    
    def emit(self, event_name, data=None):
        if event_name in self._listeners:
            event = Event(event_name, data)
            for callback in self._listeners[event_name]:
                callback(event)
    
    def off(self, event_name, callback):
        if event_name in self._listeners:
            self._listeners[event_name].remove(callback)

# 使用
emitter = EventEmitter()
emitter.on('user_created', lambda e: print(f"用户创建: {e.data}"))
emitter.emit('user_created', {'id': 1, 'name': 'Alice'})
```

## 2. 策略模式 - 支付系统

### 2.1 多种支付方式

```python
from abc import ABC, abstractmethod

class PaymentStrategy(ABC):
    @abstractmethod
    def pay(self, amount: float) -> bool:
        pass

class CreditCardPayment(PaymentStrategy):
    def __init__(self, card_number: str):
        self.card_number = card_number
    
    def pay(self, amount: float) -> bool:
        print(f"使用信用卡 {self.card_number} 支付 {amount} 元")
        return True

class AlipayPayment(PaymentStrategy):
    def pay(self, amount: float) -> bool:
        print(f"使用支付宝支付 {amount} 元")
        return True

class WeChatPayment(PaymentStrategy):
    def pay(self, amount: float) -> bool:
        print(f"使用微信支付 {amount} 元")
        return True

class PaymentContext:
    def __init__(self, strategy: PaymentStrategy):
        self.strategy = strategy
    
    def execute_payment(self, amount: float) -> bool:
        return self.strategy.pay(amount)

# 使用
payment = PaymentContext(AlipayPayment())
payment.execute_payment(100.0)
```

## 3. 装饰器模式 - 功能增强

### 3.1 日志和缓存装饰器

```python
import functools
import time
from typing import Callable

def log_execution(func: Callable) -> Callable:
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        print(f"执行函数: {func.__name__}")
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"函数 {func.__name__} 执行时间: {end - start:.2f}秒")
        return result
    return wrapper

def cache_result(func: Callable) -> Callable:
    cache = {}
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        key = str(args) + str(kwargs)
        if key in cache:
            return cache[key]
        result = func(*args, **kwargs)
        cache[key] = result
        return result
    return wrapper

@log_execution
@cache_result
def expensive_function(n: int) -> int:
    time.sleep(1)
    return n * 2
```

class OldAPI:
    def get_user_data(self, user_id: int):
        return {"id": user_id, "name": "User"}

class NewAPI:
    def fetch_user(self, identifier: int):
        return {"user_id": identifier, "username": "User"}

        self.api = api

## 5. 责任链模式 - 请求处理

### 5.1 中间件链

```python
from abc import ABC, abstractmethod

class Handler(ABC):
    def __init__(self):
        self._next = None
    
    def set_next(self, handler):
        self._next = handler
        return handler
    
    @abstractmethod
    def handle(self, request):
        if self._next:
            return self._next.handle(request)
        return None

class AuthHandler(Handler):
    def handle(self, request):
        if not request.get('token'):
            return {"error": "未认证"}
        print("认证通过")
        return super().handle(request)

class ValidationHandler(Handler):
    def handle(self, request):
        if not request.get('data'):
            return {"error": "数据无效"}
        print("验证通过")
        return super().handle(request)

class BusinessHandler(Handler):
    def handle(self, request):
        print("处理业务逻辑")
        return {"success": True, "data": request['data']}

# 构建责任链
auth = AuthHandler()
validation = ValidationHandler()
business = BusinessHandler()

auth.set_next(validation).set_next(business)

# 使用
request = {"token": "abc123", "data": "test"}
result = auth.handle(request)
```

## 6. 模板方法模式 - 算法框架

```python
from abc import ABC, abstractmethod

class DataProcessor(ABC):
    def process(self, data):
        """模板方法"""
        data = self.load(data)
        data = self.transform(data)
        data = self.validate(data)
        return self.save(data)
    
    @abstractmethod
    def load(self, data):
        pass
    
    @abstractmethod
    def transform(self, data):
        pass
    
    def validate(self, data):
        """钩子方法，可以被子类覆盖"""
        return data
    
    @abstractmethod
    def save(self, data):
        pass

class CSVProcessor(DataProcessor):
    def load(self, data):
        print("加载CSV文件")
        return data
    
    def transform(self, data):
        print("转换CSV数据")
        return data.upper()
    
    def save(self, data):
        print("保存CSV文件")
        return data

class JSONProcessor(DataProcessor):
    def load(self, data):
        print("加载JSON文件")
        return data
    
    def transform(self, data):
        print("转换JSON数据")
        return data.lower()
    
    def save(self, data):
        print("保存JSON文件")
        return data
```

## 7. 状态模式 - 状态机

### 7.1 订单状态管理

```python
from abc import ABC, abstractmethod

class OrderState(ABC):
    @abstractmethod
    def process(self, order):
        pass
    
    @abstractmethod
    def cancel(self, order):
        pass

class PendingState(OrderState):
    def process(self, order):
        print("订单处理中...")
        order.state = ProcessingState()
    
    def cancel(self, order):
        print("订单已取消")
        order.state = CancelledState()

class ProcessingState(OrderState):
    def process(self, order):
        print("订单处理完成")
        order.state = CompletedState()
    
    def cancel(self, order):
        print("订单处理中，无法取消")

class CompletedState(OrderState):
    def process(self, order):
        print("订单已完成")
    
    def cancel(self, order):
        print("订单已完成，无法取消")

class CancelledState(OrderState):
    def process(self, order):
        print("订单已取消，无法处理")
    
    def cancel(self, order):
        print("订单已取消")

class Order:
    def __init__(self):
        self.state = PendingState()
    
    def process(self):
        self.state.process(self)
    
    def cancel(self):
        self.state.cancel(self)
```

## 8. 命令模式 - 操作撤销

### 8.1 可撤销操作

```python
from abc import ABC, abstractmethod

class Command(ABC):
    @abstractmethod
    def execute(self):
        pass
    
    @abstractmethod
    def undo(self):
        pass

class AddCommand(Command):
    def __init__(self, receiver, value):
        self.receiver = receiver
        self.value = value
    
    def execute(self):
        self.receiver.add(self.value)
    
    def undo(self):
        self.receiver.subtract(self.value)

class SubtractCommand(Command):
    def __init__(self, receiver, value):
        self.receiver = receiver
        self.value = value
    
    def execute(self):
        self.receiver.subtract(self.value)
    
    def undo(self):
        self.receiver.add(self.value)

class Calculator:
    def __init__(self):
        self.value = 0
    
    def add(self, n):
        self.value += n
    
    def subtract(self, n):
        self.value -= n

class Invoker:
    def __init__(self):
        self.history = []
    
    def execute(self, command):
        command.execute()
        self.history.append(command)
    
    def undo(self):
        if self.history:
            command = self.history.pop()
            command.undo()

# 使用
calc = Calculator()
invoker = Invoker()

invoker.execute(AddCommand(calc, 10))
invoker.execute(SubtractCommand(calc, 5))
print(calc.value)  # 5

invoker.undo()
print(calc.value)  # 10
```

class Image:

        self._image = None
        self._load()

        print(f"加载图片: {self.filename}")

        self._image = None

### 12.1 FastAPI 依赖注入 - 依赖注入模式 + 工厂模式 + 作用域模式

FastAPI 的 `Depends` 是 2026 年 Python 生态中最成熟的 DI（Dependency Injection）实现，综合了依赖注入模式、工厂模式和作用域管理。

```python
"""
FastAPI 依赖注入实战：
- 依赖注入模式：路由无需手动创建依赖
- 工厂模式：每个依赖都是一个生成器/函数
- 作用域模式：request / session / app 三层生命周期
"""
from fastapi import FastAPI, Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession, create_async_engine, async_sessionmaker
from typing import Annotated, AsyncGenerator
from contextlib import asynccontextmanager

# ============ 1) App 级单例（使用 lifespan 替代 on_event） ============
@asynccontextmanager
async def lifespan(app: FastAPI):
    # 启动时创建全局资源（单例模式）
    engine = create_async_engine("postgresql+asyncpg://localhost/demo")
    app.state.db_engine = engine
    app.state.session_factory = async_sessionmaker(engine, expire_on_commit=False)
    yield
    # 关闭时清理（资源管理模式）
    await engine.dispose()


app = FastAPI(lifespan=lifespan)


# ============ 2) Request 级依赖（工厂 + 资源池） ============
async def get_db_session(
    app_state=Depends(lambda: app.state),
) -> AsyncGenerator[AsyncSession, None]:
    """每个请求独立的数据库会话（作用域模式）"""
    async with app_state.session_factory() as session:
        try:
            yield session   # 注入到路由
            await session.commit()
        except Exception:
            await session.rollback()
            raise


# ============ 3) 组合依赖：认证 + 权限 ============
async def get_current_user(
    token: str,
    db: Annotated[AsyncSession, Depends(get_db_session)],
):
    """依赖链：路由 → get_current_user → get_db_session"""
    user = await db.get(User, token)
    if not user:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED)
    return user


# ============ 4) 路由：声明式注入 ============
@app.get("/users/me")
async def read_me(
    current_user: Annotated[User, Depends(get_current_user)],
):
    # 路由完全不关心依赖如何创建
    return current_user
```

| 代码位置 | 使用的模式 |
| -------- | ---------- |
| `lifespan` | 单例模式（App 级资源）+ 资源管理模式 |
| `get_db_session` | 工厂模式 + 作用域模式（Request 级） |
| `Depends` 链 | 依赖注入模式 + 责任链 |
| 类型提示 `Annotated[T, Depends(f)]` | 声明式编程 + 类型驱动的 DI |

### 12.2 Pydantic AI Agent - 策略模式 + 类型安全的责任链

Pydantic AI（1.0.x）是 2026 年最流行的 Python Agent 框架之一，其架构本质是 "策略模式 + 类型安全的责任链"：LLM 作为可替换策略，工具作为责任链节点。

```python
"""
Pydantic AI Agent 实战：
- 策略模式：不同 LLM Provider 可互换
- 责任链：工具调用链 + 输出验证
- 依赖注入：Agent 运行时依赖
"""
from pydantic import BaseModel, Field
from pydantic_ai import Agent, RunContext
from dataclasses import dataclass


# ============ 1) 结构化输出（类型契约） ============
class TravelPlan(BaseModel):
    destination: str
    days: int = Field(ge=1, le=30)
    budget: float
    activities: list[str]


# ============ 2) Agent 运行时依赖（依赖注入） ============
@dataclass
class TravelDeps:
    user_id: str
    weather_api_key: str


# ============ 3) 创建 Agent（策略模式：指定 LLM） ============
travel_agent = Agent(
    model="openai:gpt-5.5",      # 策略可替换：anthropic:claude-opus-4-7 / google-gla:gemini-3.1
    deps_type=TravelDeps,
    output_type=TravelPlan,       # 强制结构化输出，LLM 必须返回 TravelPlan 形状
    system_prompt="你是一个专业的旅行规划助手。",
)


# ============ 4) 工具注册（责任链节点） ============
@travel_agent.tool
async def get_weather(ctx: RunContext[TravelDeps], city: str) -> dict:
    """查询城市天气（工具节点 1）"""
    # ctx.deps 就是 DI 注入的依赖
    api_key = ctx.deps.weather_api_key
    # 实际应用中这里会调用 HTTP API
    return {"city": city, "forecast": "sunny", "temp": 22}


@travel_agent.tool_plain
def search_attractions(city: str, count: int = 5) -> list[str]:
    """搜索景点（工具节点 2，无需 RunContext）"""
    return [f"{city} 景点 {i}" for i in range(count)]


# ============ 5) 输出验证（责任链中的验证环节） ============
@travel_agent.output_validator
async def validate_budget(
    ctx: RunContext[TravelDeps],
    output: TravelPlan,
) -> TravelPlan:
    """追加业务约束：预算必须合理"""
    if output.budget < output.days * 500:
        # ModelRetry 会触发 Agent 自动重新规划
        from pydantic_ai import ModelRetry
        raise ModelRetry(f"预算 {output.budget} 对于 {output.days} 天偏低")
    return output


# ============ 6) 运行 Agent ============
async def main():
    deps = TravelDeps(user_id="user_123", weather_api_key="xxx")
    result = await travel_agent.run(
        "帮我规划一次 5 天的京都旅行，预算 8000 元",
        deps=deps,
    )
    print(result.output)   # 类型是 TravelPlan，不是 str
    print(f"消耗 Token: {result.usage()}")
```

### 12.3 LLM 流式响应 - 迭代器模式 + 观察者模式

流式 LLM 输出是 2026 年常见需求，天然结合迭代器模式（异步）和观察者模式（事件驱动）：

```python
"""
LLM 流式响应实战：
- 异步迭代器模式：逐 token 消费
- 观察者模式：多订阅者（日志、指标、UI）
"""
from openai import AsyncOpenAI
from typing import AsyncIterator, Callable
import asyncio


class StreamSubject:
    """流式事件发布者（观察者模式）"""

    def __init__(self):
        self._observers: list[Callable[[str], None]] = []

    def subscribe(self, observer: Callable[[str], None]):
        self._observers.append(observer)

    def notify(self, token: str):
        for observer in self._observers:
            observer(token)


async def stream_chat(
    prompt: str,
    subject: StreamSubject,
) -> AsyncIterator[str]:
    """异步迭代器：逐 token 产出，同时通知观察者"""
    client = AsyncOpenAI()
    async with client.chat.completions.stream(
        model="gpt-5.5",
        messages=[{"role": "user", "content": prompt}],
    ) as stream:
        async for event in stream:
            if event.type == "content.delta":
                token = event.delta
                subject.notify(token)    # 观察者模式：通知所有订阅者
                yield token              # 迭代器模式：供调用方消费


# ============ 订阅者：UI / 日志 / 指标 ============
tokens_count = 0

def ui_renderer(token: str):
    print(token, end="", flush=True)

def token_counter(token: str):
    global tokens_count
    tokens_count += 1

def logger_observer(token: str):
    # 实际项目中会写日志文件或 OTel
    pass


# ============ 组合使用 ============
async def main():
    subject = StreamSubject()
    subject.subscribe(ui_renderer)
    subject.subscribe(token_counter)
    subject.subscribe(logger_observer)

    full_text = ""
    async for token in stream_chat("解释量子计算", subject):
        full_text += token    # 调用方也可以消费

    print(f"\n\n总 Token 数: {tokens_count}")
    return full_text


asyncio.run(main())
```
